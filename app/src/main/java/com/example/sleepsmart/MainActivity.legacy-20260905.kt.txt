@Composable private fun SleepSmartApp() {
    var screen by rememberSaveable { mutableStateOf(AppScreen.ANALYSIS) }
    var targetHour by rememberSaveable { mutableIntStateOf(7) }
    var targetMinute by rememberSaveable { mutableIntStateOf(30) }
    var earlyMinutes by rememberSaveable { mutableIntStateOf(30) }
    var lateMinutes by rememberSaveable { mutableIntStateOf(30) }
    var alarmEnabled by rememberSaveable { mutableStateOf(false) }
    val context = LocalContext.current
    val target = remember(targetHour, targetMinute) { targetCalendar(targetHour, targetMinute) }
    val segments = remember(targetHour, targetMinute) { VirtualSleepData.lastNight(target) }
    val result = remember(segments, earlyMinutes) { SleepAnalyzer.recommend(segments, target.time, earlyMinutes) }
    val recommended = remember(result, lateMinutes) { val latest = target.timeInMillis + lateMinutes * 60_000L; if (result.alarm.time > latest) target.time else result.alarm }
    Scaffold(containerColor = Dawn.canvas, contentWindowInsets = WindowInsets.navigationBars) { insets ->
        Column(Modifier.fillMaxSize().background(Dawn.canvas).padding(insets)) {
            Masthead(screen) { screen = it }
            when (screen) {
                AppScreen.ANALYSIS -> AnalysisScreen(result, segments) { screen = AppScreen.ALARM }
                AppScreen.ALARM -> AlarmScreen(targetHour, targetMinute, earlyMinutes, lateMinutes, result, recommended, alarmEnabled, { targetHour = it.first; targetMinute = it.second }, { targetHour = it / 60; targetMinute = it % 60 }, { earlyMinutes = it }, { lateMinutes = it }, { scheduleAlarm(context, recommended); alarmEnabled = true })
            }
        }
    }
}

@Composable private fun SleepStageChart(segments: List<SleepAnalyzer.Segment>, alarm: Date) {
    val c = LocalDawn.current
    Column(Modifier.semantics(mergeDescendants = true) { contentDescription = "睡眠阶段时间轴，推荐唤醒时间 ${formatClock(alarm)}" }) {
        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) { Text(SimpleDateFormat("HH:mm", Locale.getDefault()).format(segments.first().start), color = c.muted, style = MaterialTheme.typography.labelSmall); Text("推荐 ${formatClock(alarm)}", color = c.sage, style = MaterialTheme.typography.labelSmall, fontWeight = FontWeight.Bold); Text(SimpleDateFormat("HH:mm", Locale.getDefault()).format(segments.last().end), color = c.muted, style = MaterialTheme.typography.labelSmall) }
        Spacer(Modifier.height(8.dp))
        Canvas(Modifier.fillMaxWidth().height(132.dp)) {
            val total = segments.sumOf { (it.end.time - it.start.time).coerceAtLeast(0L) }.toFloat().coerceAtLeast(1f); var x = 0f; val baseline = size.height * .82f
            drawLine(c.rule, Offset(0f, baseline), Offset(size.width, baseline), 1.dp.toPx()); repeat(7) { tick -> val tx = size.width * tick / 6f; drawLine(c.rule.copy(alpha = .6f), Offset(tx, 0f), Offset(tx, baseline), 1.dp.toPx()) }
            segments.forEach { segment -> val width = size.width * ((segment.end.time - segment.start.time).coerceAtLeast(0L) / total); val top = when (segment.stage) { SleepAnalyzer.Stage.DEEP -> size.height * .15f; SleepAnalyzer.Stage.REM -> size.height * .34f; SleepAnalyzer.Stage.LIGHT -> size.height * .51f; else -> size.height * .67f }; val color = when (segment.stage) { SleepAnalyzer.Stage.DEEP -> c.deep; SleepAnalyzer.Stage.REM -> c.rem; SleepAnalyzer.Stage.LIGHT -> c.light; else -> c.muted }; drawRect(color, Offset(x, top), Size(width.coerceAtLeast(2f), baseline - top)); x += width }
            val wakeFraction = (alarm.time - segments.first().start.time).toFloat() / (segments.last().end.time - segments.first().start.time).coerceAtLeast(1L); val wakeX = (size.width * wakeFraction).coerceIn(0f, size.width); drawLine(c.sage, Offset(wakeX, 0f), Offset(wakeX, baseline + 8.dp.toPx()), 2.dp.toPx(), cap = StrokeCap.Round); drawCircle(c.sage, 4.dp.toPx(), Offset(wakeX, baseline))
        }
        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(17.dp)) { ChartKey("深睡", c.deep); ChartKey("浅睡", c.light); ChartKey("REM", c.rem) }
    }
}

@Composable private fun DialClock(targetTotal: Int, early: Int, late: Int, onTarget: (Int) -> Unit, onEarly: (Int) -> Unit, onLate: (Int) -> Unit) {
    val c = LocalDawn.current; val density = LocalDensity.current; val dialDp = minOf(330.dp, 1000.dp); val dialPx = with(density) { dialDp.toPx() }; val start = normalizedMinutes(targetTotal - early); val end = normalizedMinutes(targetTotal + late)
    Box(Modifier.fillMaxWidth(), contentAlignment = Alignment.Center) {
        Canvas(Modifier.size(dialDp).semantics { contentDescription = "圆盘闹钟，目标 ${formatTimeFromMinutes(targetTotal)}，提前 ${early} 分钟，延后 ${late} 分钟" }.pointerInput(targetTotal, early, late) {
            var active = "target"
            detectDragGestures(onDragStart = { point ->
                val center = Offset(dialPx / 2f, dialPx / 2f); val threshold = with(density) { 39.dp.toPx() }; val candidates = listOf("start" to pointForMinutes(start, center, dialPx * .36f), "target" to pointForMinutes(targetTotal, center, dialPx * .36f), "end" to pointForMinutes(end, center, dialPx * .36f)); active = candidates.minBy { distance(it.second, point) }.let { if (distance(it.second, point) <= threshold) it.first else "target" }
            }, onDrag = { change, _ ->
                val center = Offset(dialPx / 2f, dialPx / 2f); val minute = minuteForPoint(change.position, center); when (active) { "start" -> onEarly(((targetTotal - minute + 1440) % 1440).coerceIn(5, 90)); "end" -> onLate(((minute - targetTotal + 1440) % 1440).coerceIn(5, 90)); else -> onTarget(minute) }; change.consume()
            })
        }) {
            val center = Offset(size.width / 2f, size.height / 2f); val radius = size.minDimension * .41f
            drawCircle(c.rule, radius, center, style = Stroke(width = 1.dp.toPx()))
            val windowStartAngle = start / 1440f * 360f - 90f
            val windowSweep = (early + late).coerceAtMost(720) / 1440f * 360f
            drawArc(c.sage.copy(alpha = .18f), windowStartAngle, windowSweep, false, center - Offset(radius, radius), Size(radius * 2, radius * 2), style = Stroke(18.dp.toPx(), cap = StrokeCap.Round))
            for (tick in 0 until 24) { val angle = tick * 2f * PI.toFloat() / 24f - PI.toFloat() / 2f; val outer = pointForAngle(angle, center, radius + 7.dp.toPx()); val inner = pointForAngle(angle, center, radius - if (tick % 6 == 0) 14.dp.toPx() else 7.dp.toPx()); drawLine(if (tick % 6 == 0) c.text else c.rule, inner, outer, if (tick % 6 == 0) 2.dp.toPx() else 1.dp.toPx(), cap = StrokeCap.Round) }
            listOf(start to c.light, targetTotal to c.sage, end to c.rem).forEachIndexed { index, item -> val p = pointForMinutes(item.first, center, radius); val inner = pointForMinutes(item.first, center, radius * .55f); drawLine(item.second, center, inner, if (index == 1) 3.dp.toPx() else 2.dp.toPx(), cap = StrokeCap.Round); drawLine(item.second, inner, p, if (index == 1) 3.dp.toPx() else 2.dp.toPx(), cap = StrokeCap.Round); if (index == 1) { drawCircle(item.second, 9.dp.toPx(), p); drawCircle(c.canvas, 4.dp.toPx(), p) } else { drawCircle(c.canvas, 9.dp.toPx(), p); drawCircle(item.second, 9.dp.toPx(), p, style = Stroke(2.dp.toPx())) } }
            drawCircle(c.text, 3.dp.toPx(), center)
        }
        Text("00", color = c.muted, style = MaterialTheme.typography.labelSmall, modifier = Modifier.align(Alignment.TopCenter).padding(top = 7.dp))
        Text("06", color = c.muted, style = MaterialTheme.typography.labelSmall, modifier = Modifier.align(Alignment.BottomCenter).padding(bottom = 7.dp))
        Text("12", color = c.muted, style = MaterialTheme.typography.labelSmall, modifier = Modifier.align(Alignment.CenterStart).padding(start = 7.dp))
        Text("18", color = c.muted, style = MaterialTheme.typography.labelSmall, modifier = Modifier.align(Alignment.CenterEnd).padding(end = 7.dp))
        Column(horizontalAlignment = Alignment.CenterHorizontally) { Text(formatTimeFromMinutes(targetTotal), fontSize = 39.sp, fontWeight = FontWeight.Bold); Text("目标起床", color = c.muted, style = MaterialTheme.typography.labelSmall) }
    }
}

@Composable private fun RangeScrubber(early: Int, late: Int, onEarly: (Int) -> Unit, onLate: (Int) -> Unit) {
    val c = LocalDawn.current
    Row(verticalAlignment = Alignment.CenterVertically) { Text("−", color = c.sage, fontSize = 23.sp, modifier = Modifier.width(21.dp), textAlign = TextAlign.Center); Slider(value = (early + late).toFloat(), onValueChange = { val total = it.toInt().coerceIn(10, 180); val left = (total * early.toFloat() / (early + late).coerceAtLeast(1)).toInt().coerceIn(5, 90); onEarly(left); onLate((total - left).coerceIn(5, 90)) }, valueRange = 10f..180f, steps = 16, modifier = Modifier.weight(1f)); Text("+", color = c.sage, fontSize = 23.sp, modifier = Modifier.width(21.dp), textAlign = TextAlign.Center) }
}

private fun pointForAngle(angle: Float, center: Offset, radius: Float) = Offset(center.x + cos(angle) * radius, center.y + sin(angle) * radius)
private fun pointForMinutes(minutes: Int, center: Offset, radius: Float): Offset = pointForAngle(minutes / 1440f * 2f * PI.toFloat() - PI.toFloat() / 2f, center, radius)
private fun minuteForPoint(point: Offset, center: Offset): Int { val angle = atan2(point.y - center.y, point.x - center.x) + PI.toFloat() / 2f; val normalized = (angle + 2f * PI.toFloat()) % (2f * PI.toFloat()); return ((normalized / (2f * PI.toFloat())) * 1440f).toInt() / 5 * 5 }
private fun distance(a: Offset, b: Offset) = sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y))
private fun normalizedMinutes(minutes: Int) = ((minutes % 1440) + 1440) % 1440
private fun targetCalendar(hour: Int, minute: Int): Calendar = Calendar.getInstance().apply { set(Calendar.HOUR_OF_DAY, hour); set(Calendar.MINUTE, minute); set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0); if (timeInMillis <= System.currentTimeMillis() + 5 * 60_000L) add(Calendar.DAY_OF_YEAR, 1) }
private fun formatTimeFromMinutes(minutes: Int) = formatTime(minutes / 60, minutes % 60)
private fun formatTime(hour: Int, minute: Int): String { val total = normalizedMinutes(hour * 60 + minute); return String.format(Locale.getDefault(), "%02d:%02d", total / 60, total % 60) }
private fun formatClock(date: Date) = SimpleDateFormat("HH:mm", Locale.getDefault()).format(date)
private fun formatDuration(minutes: Long) = String.format(Locale.getDefault(), "%dh %02dm", minutes / 60, minutes % 60)

private fun scheduleAlarm(context: Context, date: Date) {
    val manager = context.getSystemService(AlarmManager::class.java); val intent = Intent(context, WakeReceiver::class.java); val pending = PendingIntent.getBroadcast(context, 1001, intent, PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)
    if (Build.VERSION.SDK_INT >= 31 && !manager.canScheduleExactAlarms()) { context.startActivity(Intent(android.provider.Settings.ACTION_REQUEST_SCHEDULE_EXACT_ALARM)); Toast.makeText(context, "请允许精确闹钟权限后再开启", Toast.LENGTH_LONG).show(); return }
    manager.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, date.time, pending); Toast.makeText(context, "智能闹钟已设置为 ${formatClock(date)}", Toast.LENGTH_SHORT).show()
}
